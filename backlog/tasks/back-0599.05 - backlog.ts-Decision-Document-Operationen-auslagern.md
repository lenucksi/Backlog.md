```typescript
await this._filesystem.saveDecision(decision);

  await this.git.stageBacklogDirectory();
  await this.git.commitChanges("...");
}
```
Wird zu:
```typescript
export async function createDecision(
  deps: { filesystem: FileSystem; git: GitOperations },
  decision: Decision,
  autoCommit?: boolean,
): Promise<void> {
  await deps.filesystem.saveDecision(decision);
  await withGitCommit(deps, autoCommit, "Create decision ...");
}
```
